148 template<typename Tg, typename ReqPtr>
149 void call(
150 ReqPtr req,
151 std::function<void(typename call_result_ptr<Tg>::type)> result_callback,
152 std::function<void(typename call_error_ptr<Tg>::type)> error_callback
153 ) const
154 {
155 typedef std::function<void(
156 typename call_result_ptr<Tg>::type,
157 typename call_error_ptr<Tg>::type
158 )> callback_type;
159
160 using namespace std::placeholders;
161 callback_type callback = nullptr;
162 if ( result_callback!=nullptr )
163 callback = std::bind( self::response_handler<Tg>, _1, _2, result_callback, error_callback );
164
165 this->get_aspect().template get<Tg>().call(
166 *this,
167 std::move(req),
168 std::move(callback)
169 );
170 }

222 template<typename Schema, typename L, typename R>
223 static void create_schema_(std::vector<Schema>* sch_list, const std::vector<std::string>& names, fas::type_list<L, R>)
224 {
225 typedef L head_tag;
226 typedef R tail_type;
227 typedef typename super::aspect::template advice_cast<head_tag>::type method_type;
228 sch_list->push_back( method_type::template create_schema_t<Schema>(names) );
229 create_schema_(sch_list, names, tail_type() );
